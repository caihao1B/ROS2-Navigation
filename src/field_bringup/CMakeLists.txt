function(simulation_plugin name file)
  add_library(${name} SHARED src/${file})
  target_link_libraries(${name} gz-sim8::gz-sim8 gz-plugin2::gz-plugin2 gz-transport13::gz-transport13 nlohmann_json::nlohmann_json)
  install(TARGETS ${name} LIBRARY DESTINATION lib)
endfunction()
