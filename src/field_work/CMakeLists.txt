cmake_minimum_required(VERSION 3.16)
project(field_work)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(field_interfaces REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(std_msgs REQUIRED)
find_package(rosgraph_msgs REQUIRED)
find_package(nav2_regulated_pure_pursuit_controller REQUIRED)
find_package(nav2_core REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(pluginlib REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz_transport_vendor REQUIRED)
# Gazebo on Noble links the distro protobuf ABI. Prefer its matching compiler
# if a separate ROS dependency has prepended a different protoc to PATH.
find_program(SYSTEM_PROTOC protoc PATHS /usr/bin NO_DEFAULT_PATH)
if(SYSTEM_PROTOC AND NOT DEFINED Protobuf_PROTOC_EXECUTABLE)
  set(Protobuf_PROTOC_EXECUTABLE "${SYSTEM_PROTOC}" CACHE FILEPATH "Gazebo-compatible protobuf compiler")
endif()
find_package(Protobuf MODULE REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
find_package(gz-transport13 REQUIRED)
find_package(nlohmann_json REQUIRED)

add_library(directed_path_curvature SHARED src/directed_path_controller.cpp)
ament_target_dependencies(directed_path_curvature nav2_regulated_pure_pursuit_controller
  nav2_core nav2_costmap_2d pluginlib tf2 tf2_geometry_msgs)
pluginlib_export_plugin_description_file(nav2_core plugins.xml)
add_library(field_tool_actuator SHARED src/tool_actuator.cpp)
target_link_libraries(field_tool_actuator gz-sim8::gz-sim8 gz-plugin2::gz-plugin2
  gz-transport13::gz-transport13 nlohmann_json::nlohmann_json)
add_executable(tool_action_server src/tool_action_server.cpp)
ament_target_dependencies(tool_action_server field_interfaces rclcpp rclcpp_action std_msgs rosgraph_msgs)
target_link_libraries(tool_action_server nlohmann_json::nlohmann_json)

ament_python_install_package(${PROJECT_NAME})
install(TARGETS directed_path_curvature field_tool_actuator LIBRARY DESTINATION lib)
install(TARGETS tool_action_server RUNTIME DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/work_ledger DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(work_semantics test/test_work_semantics.py)
  find_package(ament_cmake_test REQUIRED)
  add_executable(controller_registration test/controller_registration.cpp)
  ament_target_dependencies(controller_registration pluginlib nav2_core)
  ament_add_test(controller_registration COMMAND $<TARGET_FILE:controller_registration>
    GENERATE_RESULT_FOR_RETURN_CODE_ZERO)
endif()
ament_package()
